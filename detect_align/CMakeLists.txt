cmake_minimum_required(VERSION 3.15)
project(detect_align LANGUAGES C CXX)
if(NOT DEFINED MODEL_ZOO)
    message(FATAL_ERROR "Set MODEL_ZOO to the rknn_model_zoo directory")
endif()
set(TARGET_SOC rk356x)
set(DISABLE_RGA ON)
add_subdirectory(${MODEL_ZOO}/3rdparty 3rdparty.out)
add_subdirectory(${MODEL_ZOO}/utils utils.out)
add_executable(detect_face main.cc ../third_party/retinaface/rknpu2/retinaface.cc)
target_include_directories(detect_face PRIVATE ../third_party/retinaface ${LIBRKNNRT_INCLUDES})
target_link_libraries(detect_face imageutils fileutils ${LIBRKNNRT} dl m)
set_target_properties(detect_face PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
add_executable(face_align face_align.c)
target_link_libraries(face_align m)
add_executable(sface_extract sface_extract.c)
target_include_directories(sface_extract PRIVATE ${LIBRKNNRT_INCLUDES})
target_link_libraries(sface_extract ${LIBRKNNRT} m)
set_target_properties(sface_extract PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
add_executable(compare_feature compare_feature.c)
target_link_libraries(compare_feature m)
install(TARGETS detect_face face_align sface_extract compare_feature RUNTIME DESTINATION .)
install(PROGRAMS recognize_once.sh DESTINATION .)
